cmake_minimum_required(VERSION 3.24)
project(GIGPBenchmark LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(GIGP_CPU_ARCH "native" CACHE STRING "CPU architecture for this allocated worker")
option(GIGP_RAFT_TEMPLATE_REDUCE "Use RAFT's templated layout/direction API" OFF)
get_filename_component(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP REQUIRED)
find_package(spdlog REQUIRED)
find_package(TBB REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Python COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 REQUIRED)
find_package(raft REQUIRED)
find_library(OPENBLAS_LIBRARY NAMES openblas libopenblas.so.0 REQUIRED)
add_library(cnpy STATIC "${UPSTREAM}/deps/cnpy/cnpy.cpp")
target_include_directories(cnpy PUBLIC "${UPSTREAM}/deps/cnpy")
target_link_libraries(cnpy PUBLIC ZLIB::ZLIB)
add_executable(IGPGPUPP "${UPSTREAM}/src/app/IGPGPUPP.cpp")
set_source_files_properties("${UPSTREAM}/src/app/IGPGPUPP.cpp" PROPERTIES LANGUAGE CUDA)
target_include_directories(IGPGPUPP PRIVATE "${UPSTREAM}/src" "${UPSTREAM}/deps/cmd_line_parser/include")
target_link_libraries(IGPGPUPP PRIVATE cnpy OpenMP::OpenMP_CXX CUDA::cublas CUDA::cublasLt spdlog::spdlog TBB::tbb pybind11::embed raft::raft ${OPENBLAS_LIBRARY})
target_compile_definitions(IGPGPUPP PRIVATE USE_GPU USE_RETRIEVAL_GPU USE_AVX512 LIBCUDACXX_ENABLE_EXPERIMENTAL_MEMORY_RESOURCE)
if(GIGP_RAFT_TEMPLATE_REDUCE)
  target_compile_definitions(IGPGPUPP PRIVATE GIGP_RAFT_TEMPLATE_REDUCE)
endif()
target_compile_options(IGPGPUPP PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-fopenmp,-march=${GIGP_CPU_ARCH};--expt-extended-lambda;--expt-relaxed-constexpr;-O3>)
add_executable(prepare_ivf prepare_ivf.cpp)
target_include_directories(prepare_ivf PRIVATE "${UPSTREAM}/src")
target_link_libraries(prepare_ivf PRIVATE cnpy spdlog::spdlog)
add_executable(raft_reduce_check test_raft_reduce.cu)
target_include_directories(raft_reduce_check PRIVATE "${UPSTREAM}/src")
target_link_libraries(raft_reduce_check PRIVATE raft::raft CUDA::cudart)
target_compile_options(raft_reduce_check PRIVATE --expt-extended-lambda --expt-relaxed-constexpr)
if(GIGP_RAFT_TEMPLATE_REDUCE)
  target_compile_definitions(raft_reduce_check PRIVATE GIGP_RAFT_TEMPLATE_REDUCE)
endif()
